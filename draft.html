# ==============================================================================
# 1. PROCESS DATA (Directly from 'x')
# ==============================================================================
# We assume 'x' is your raw loaded dataframe.

plot_data <- x %>%
  # A. Determine First Pitch Result (Ball vs Strike)
  mutate(
    start_type = case_when(
      grepl("ball|hit_by_pitch|blocked", tolower(first_pitch_result)) ~ "Ball",
      TRUE ~ "Strike"
    )
  ) %>%
  
  # B. Calculate Stats
  group_by(player_name, start_type) %>%
  summarize(
    K_pct = mean(is_strikeout, na.rm = TRUE),
    batters = n(),
    .groups = "drop"
  ) %>%
  pivot_wider(names_from = start_type, values_from = c(K_pct, batters)) %>%
  
  # C. Filter Valid Data
  filter(!is.na(K_pct_Ball) & !is.na(K_pct_Strike)) %>%
  mutate(
    total_batters = batters_Ball + batters_Strike,
    K_Diff = K_pct_Strike - K_pct_Ball
  ) %>%
  filter(total_batters >= 50) %>% # Minimum sample size
  
  # D. Define Roles (Starter, Reliever, Closer)
  mutate(
    detailed_role = case_when(
      total_batters > 150 ~ "Starter",
      total_batters <= 150 & K_pct_Strike > 0.30 ~ "Closer",
      TRUE ~ "Reliever"
    )
  ) %>%
  
  # E. Keep Top 15 Only
  group_by(detailed_role) %>%
  arrange(desc(K_Diff)) %>%
  slice_head(n = 15) %>%
  ungroup() %>%
  
  # F. Sort for Plotting
  mutate(player_name = fct_reorder(player_name, K_Diff))
